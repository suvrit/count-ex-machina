\cxtitle{Log-concavity of all Derivatives of the Jacobi-theta kernel}

\begin{cxcredits}

\begin{cxcontext}
Write $\xi(s)=\tfrac12 s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s)$ for the Riemann $\xi$-function and $\Xi(x)=\xi(\tfrac12+ix)$ for its restriction to the critical line, a real even entire function of order one. Pólya's Fourier-cosine representation of $\Xi$ (see \citealp[\S4]{Csordas2014}, who writes $H(x)$ for $\Xi(x/2)/8$) is
\[
\Xi(x/2)=8\int_0^\infty \Phi(t)\cos(xt)\,dt,
\]
where $\Phi$ is the kernel displayed in the statement. The Riemann Hypothesis is equivalent to all zeros of $\Xi$ being real, that is, to $\Xi$ belonging to the Laguerre--P\'olya class of real entire functions that are locally uniform limits of real polynomials with only real zeros. Nothing is known that characterizes, in terms of a kernel, when its cosine transform has only real zeros; the program behind the statement below is to find properties of $\Phi$ that constrain the zeros of $\Xi$.

Classical facts about the kernel (P\'olya, Wintner; collected as Theorem~4.1 of \citealp{Csordas2014}): $\Phi$ is even and positive, analytic in the strip $|\operatorname{Im}z|<\pi/8$, satisfies $\Phi'(t)<0$ for $t>0$, and every derivative $\Phi^{(n)}(t)$ decays faster than $\exp(-(\pi-\varepsilon)e^{4t})$ as $t\to\infty$. Because $\Phi$ is even, each $J_n$ below is even, so the conjecture is a statement about $t\ge0$.

The interest in derivatives of $\Phi$ comes from the Tur\'an inequalities. The Maclaurin coefficients of $\Xi$ are, up to the normalization, the moments $b_k=\int_0^\infty t^{2k}\Phi(t)\,dt$, since $\Xi(x/2)=8\sum_{k\ge0}(-1)^k b_k\,x^{2k}/(2k)!$, and a necessary condition for the Riemann Hypothesis (P\'olya--Schur) is
\[
b_k^2-\tfrac{2k-1}{2k+1}\,b_{k-1}b_{k+1}\ge0\qquad(k\ge1).
\]
These were proved by Csordas, Norfolk and Varga \citeyearpar{CsordasNorfolkVarga1986} and by Csordas and Varga \citeyearpar{CsordasVarga1988} from concavity properties of $\Phi$, in particular the strict concavity of $\log\Phi(\sqrt t\,)$ on $t>0$, which is stronger than the log-concavity of $\Phi$ itself. Coffey and Csordas \citeyearpar{coffey2013log} gave a new proof that $\Phi$ is log-concave, the case $n=1$ below, and conjectured that every derivative of $\Phi$ is strictly log-concave as well; Csordas restated that conjecture as Open Problem~4.13.

A positive function $g$ is \emph{strictly log-concave} when $(\log g)''<0$, equivalently $(g')^2-gg''>0$. For $n\ge2$ the derivative $\Phi^{(n-1)}$ need not be positive (already $\Phi'\le0$ on $t\ge0$), and the source takes the inequality $J_n(t)>0$ itself as the meaning of strict log-concavity of $\Phi^{(n-1)}$; that inequality is what is refuted below.
\end{cxcontext}

\begin{cxsource}{theta-derivative-log-concavity}

\end{cxsource}

\begin{cxstatement}{theta-derivative-log-concavity}
([5, Conjecture 2.5])  Show that the derivatives of the Jacobi theta function, $\Phi(t)$, are (strictly) log-concave on $\R$; that is, for each $n\in\mathbb N$,
\[J_n(t):=(\Phi^{(n)}(t))^2-\Phi^{(n-1)}(t)\Phi^{(n+1)}(t)>0\quad\text{for }t\in\R.\]
Here, by (4.2) of the same paper, the Jacobi theta function (without the usual factor $4$) is $\Phi(t):=\sum_{n\ge1}\pi n^2\left(2\pi n^2e^{4t}-3\right)\exp\left(5t-\pi n^2e^{4t}\right)$.
\end{cxstatement}

\begin{cxsummary}{theta-derivative-log-concavity}

\end{cxsummary}

\begin{cxcertificate}{theta-derivative-log-concavity}
Rigorous interval evaluation of $J_9(1/50)<0$.
\end{cxcertificate}

\begin{cxrefutation}
\begin{theorem}[\statusfalse; computer-assisted proof]\label{thm:theta}
At $n=9$ and $t=1/50$, one has $J_9(t)<0$.
\end{theorem}
\begin{proof}
Put $u=\pi m^2e^{4t}$ and define integer polynomials
\[
P_0(u)=2u-3,\qquad P_{k+1}(u)=4uP_k'(u)+(5-4u)P_k(u).
\]
Termwise differentiation gives
\[
\Phi^{(k)}(t)=\sum_{m\ge1}\pi m^2e^{5t-u}P_k(u).
\]
Outward-rounded interval summation, with an explicit geometrically bounded tail, yields
\begin{align*}
\Phi^{(8)}(1/50)&\approx 2.95940368409056694\cdot10^8,\\
\Phi^{(9)}(1/50)&\approx 5.80442287530267756\cdot10^9,\\
\Phi^{(10)}(1/50)&\approx 2.02040519425950258\cdot10^{11},
\end{align*}
and an interval for $J_9(1/50)$ contained strictly in the negative half-line, centered at
\[
-2.61006208371358922\cdot10^{19}.
\]
The source package contains a high-precision evaluator and a Sage \texttt{RealIntervalField} certificate.  The negative margin is many orders of magnitude larger than the certified truncation error.
\end{proof}

\begin{remark}
  GPT-5.5 managed to prove the Tur\'an inequalities for $\Phi^{(n)}$ for $n=2$ and $n=3$; while we were attempting an inductive proof to generalize those proofs to all $n$ is when the counterexample popped up. However, we have not yet verified whether the Tur\'an inequalities hold for all $n \le 8$.
\end{remark}
\end{cxrefutation}
